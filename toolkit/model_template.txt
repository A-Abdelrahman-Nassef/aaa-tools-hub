function bbox(p){ const xs=p.map(q=>q[0]), ys=p.map(q=>q[1]); return [Math.min(...xs),Math.min(...ys),Math.max(...xs),Math.max(...ys)]; }
function inside(p, x, y){ let c=false; for(let i=0,j=p.length-1;i<p.length;j=i++){ const [xi,yi]=p[i],[xj,yj]=p[j]; if(((yi>y)!=(yj>y)) && (x < (xj-xi)*(y-yi)/(yj-yi)+xi)) c=!c; } return c; }
function labelPt(p){ // a point inside the polygon near its centroid

function stairs(group, r, z0, h, key){
  const [x0,y0,x1,y1] = bbox(r), w = x1-x0, L = y1-y0;
  // flights run along X (longer side)
  const along = w > L, run = (along ? w : L) - 1.1, n = 9, top = h - 0.3;
  for(let f=0; f<2; f++) for(let i=0;i<n;i++){
    const zt = z0 + 0.09 + f*top/2 + (top/2)*(i+1)/n;
    if(along){
      const xa = f===0 ? x0 + run*i/n : x1 - 1.1 - run*(i+1)/n, xb = f===0 ? x0 + run*(i+1)/n : x1 - 1.1 - run*i/n;
      const ya = f===0 ? y0+.05 : y0+L/2+.05, yb = f===0 ? y0+L/2-.05 : y1-.05;
      addItem(group, boxMesh(xa,ya,xb,yb,zt-.17,zt), key, 'step');
    }
  }
  addItem(group, boxMesh(x1-1.1, y0+.05, x1-.05, y1-.05, z0+.09+top/2-.17, z0+.09+top/2), key, 'step');
}
function slats(group, poly, zb, zt, key, bb){
  const b = bbox(poly);
  for(let y = b[1]+0.15; y < b[3]; y += 0.32){
    const xs = [];
    for(let i=0,j=poly.length-1;i<poly.length;j=i++){ const [xi,yi]=poly[i],[xj,yj]=poly[j]; if((yi>y)!=(yj>y)) xs.push(xi + (y-yi)*(xj-xi)/(yj-yi)); }
    xs.sort((a,b)=>a-b);
    for(let k=0;k+1<xs.length;k+=2) addItem(group, boxMesh(xs[k], y-0.04, xs[k+1], y+0.04, zb, zt), key, 'canopy', '#C9CED4');
  }
  // perimeter beams
  for(let i=0;i<poly.length;i++){ const [a,b2]=[poly[i], poly[(i+1)%poly.length]];
    const x0=Math.min(a[0],b2[0]), x1=Math.max(a[0],b2[0]), y0=Math.min(a[1],b2[1]), y1=Math.max(a[1],b2[1]);
    const onBuilding = bb && (x0>=bb[0]-.01 && x1<=bb[2]+.01 && y0>=bb[1]-.01 && y1<=bb[3]+.01);
    if(onBuilding) continue;
    addItem(group, boxMesh(x0-.05, y0-.05, Math.max(x1,x0+.1)+.05, Math.max(y1,y0+.1)+.05, zb-.12, zt), key, 'canopy', '#B9C0C7');
    // a post at the outer corner
    addItem(group, boxMesh(a[0]-.08, a[1]-.08, a[0]+.08, a[1]+.08, 0, zb-.12), key, 'detail', '#AEB5BC');
  }
}
function parapet(group, poly, z0, z1, key){
  for(let i=0;i<poly.length;i++){ const [a,b]=[poly[i], poly[(i+1)%poly.length]];
    const x0=Math.min(a[0],b[0]), x1=Math.max(a[0],b[0]), y0=Math.min(a[1],b[1]), y1=Math.max(a[1],b[1]);
    addItem(group, boxMesh(x0, y0, x1===x0 ? x0+.15 : x1, y1===y0 ? y0+.15 : y1, z0, z1), key, 'struct', '#C3C8CE');
  }
}

function stateOf(k){

        else { set(tok.ghost, .45, true); if(m.map){ m.map = null; m.needsUpdate = true; } } break;

    }
  });

}

new MutationObserver(retheme).observe(document.documentElement, {attributes:true, attributeFilter:['data-theme']});
const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
const v = new THREE.Vector3();
window.__villa = window.__model = {setMode, setLevel, setPlan, select, get explode(){return explode;}, set explodeT(x){explodeT = x;}, camera, controls, render:() => renderer.render(scene, camera), levelGroups};

  const prev = explode;
  explode += reduce ? (explodeT - explode) : (explodeT - explode) * 0.12; if(Math.abs(explodeT - explode) < 0.001) explode = explodeT;
  levelGroups.forEach((L, i) => L.group.position.y = i * GAP * explode);
  if(prev !== explode){ const dy = (explode - prev) * GAP * (level === 'all' ? 1 : levelGroups.indexOf(LVG(level))); controls.target.y += dy; camera.position.y += dy; }
  controls.update(); renderer.render(scene, camera);
  const w = stage.clientWidth, h = stage.clientHeight;
  const project = (p, el) => { v.copy(p).project(camera); const vis = v.z < 1 && v.z > -1 && Math.abs(v.x) < 1.05 && Math.abs(v.y) < 1.05; el.hidden = !vis; if(vis){ el.style.left = ((v.x+1)/2*w)+'px'; el.style.top = ((1-v.y)/2*h)+'px'; } };
  levelGroups.forEach(L => { if(!L.group.visible || level !== 'all'){ L.labelEl.hidden = true; return; } const a = L.anchor.clone(); a.y += L.group.position.y; project(a, L.labelEl); });
  roomLabels.forEach(rl => { if(level !== rl.level){ rl.el.hidden = true; return; } const p = rl.local.clone(); p.y += rl.group.position.y; project(p, rl.el); });
  requestAnimationFrame(frame);
}
levelGroups.forEach((L, i) => L.group.position.y = i * GAP);
frameView(); sync(); frame();
})();
